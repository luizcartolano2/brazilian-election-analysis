WITH validos AS (
  SELECT turno, uf, municipio, sum(votos) AS validos
  FROM votos_municipio
  WHERE cargo = 1 AND tipo IN (1, 2)
  GROUP BY turno, uf, municipio
)
SELECT v.uf, m.nome AS municipio,
       sum(v.votos) AS votos,
       ((sum(v.votos)::BIGINT * 20000 + d.validos)
         // (2 * d.validos)) / 100 AS percentual_validos
FROM votos_municipio v
JOIN validos d USING (turno, uf, municipio)
JOIN municipios m ON m.municipio = v.municipio
WHERE v.cargo = 1 AND v.tipo = 1 AND v.numero = 13
GROUP BY v.turno, v.uf, v.municipio, m.nome, d.validos
ORDER BY votos DESC
LIMIT 100
